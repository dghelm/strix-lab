cmake_minimum_required(VERSION 3.20)
project(strixlab_topk_reference LANGUAGES C CXX)
# The fixed native build adapter supplies and authenticates these cache entries.
set(STRIXLAB_NATIVE_BUILD_COMMIT "unbound-host-test" CACHE STRING "Authenticated source version")
set(STRIXLAB_NATIVE_BUILD_NUMBER "0" CACHE STRING "Native source build number")
if(NOT STRIXLAB_NATIVE_BUILD_NUMBER STREQUAL "0")
  message(FATAL_ERROR "STRIXLAB_NATIVE_BUILD_NUMBER must be 0")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(topk_reference STATIC reference.cpp)
target_compile_features(topk_reference PUBLIC cxx_std_17)
target_include_directories(topk_reference PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(topk_reference PRIVATE OpenSSL::Crypto)
target_compile_options(topk_reference PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(topk_host_fixture STATIC host_fixture.cpp capsule_transport.cpp)
target_link_libraries(topk_host_fixture PUBLIC topk_reference PRIVATE OpenSSL::Crypto)
target_compile_options(topk_host_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(topk_capsule_host_test capsule_main.cpp)
target_link_libraries(topk_capsule_host_test PRIVATE topk_host_fixture)
add_executable(topk_reference_test ../../tests/native/topk_reference_test.cpp)
target_link_libraries(topk_reference_test PRIVATE topk_reference)
add_executable(topk_capsule_test ../../tests/native/topk_capsule_test.cpp)
target_link_libraries(topk_capsule_test PRIVATE topk_host_fixture)
add_executable(topk_baseline_network_test ../../tests/native/topk_baseline_network_test.cpp)
option(STRIXLAB_TOPK_BUILD_FAULT_FIXTURES "Build compile-time transport fault fixtures" OFF)
if(STRIXLAB_TOPK_BUILD_FAULT_FIXTURES)
  foreach(fault boundary_tie zero_sign duplicate index count order nan replay_order benchmark_unready)
    add_executable(topk_capsule_fault_${fault} capsule_main.cpp)
    target_compile_definitions(topk_capsule_fault_${fault} PRIVATE STRIXLAB_TOPK_TEST_FAULT=${fault})
    target_link_libraries(topk_capsule_fault_${fault} PRIVATE topk_host_fixture)
  endforeach()
endif()
enable_testing()
add_test(NAME topk_reference COMMAND topk_reference_test)
add_test(NAME topk_capsule_gate COMMAND topk_capsule_test)
add_test(NAME topk_baseline_network COMMAND topk_baseline_network_test)
